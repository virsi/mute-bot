# News Digest Bot — Phase 1 (Solo-mode MVP) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the single-user MVP of the news digest bot: read configured Telegram channels via MTProto, deduplicate posts into clusters, classify by preset topics, rank by importance, deliver scheduled digests via Bot API. No payments, no alerts, no custom topics — those land in Phases 2–3.

**Architecture:** Three runtime processes (`session-reader`, `processor`, `bot-api`) plus a `scheduler`. They communicate over NATS JetStream and share Postgres (with pgvector) and Redis. Workers are stateless and idempotent. The `processor` binary hosts the normalizer, dedup-worker, classifier, ranker and digest-assembler as separate JetStream consumers inside one process for Phase 1; later phases can split them out.

**Tech Stack:** Go 1.23+, Postgres 16 + pgvector, Redis 7, NATS JetStream 2.10+, `gotd/td` (MTProto), `go-telegram/bot` (Bot API), `pgx/v5`, `redis/go-redis/v9`, `nats.go`, `gocron/v2`, `golang-migrate`, `slog`, `prometheus/client_golang`, `otel`, `testify`, `testcontainers-go`, OpenAI Go SDK.

**Spec reference:** `docs/superpowers/specs/2026-06-03-news-digest-bot-design.md`

## Milestone Overview

| # | Milestone | Outcome |
|---|---|---|
| M1 | Foundation | Repo skeleton compiles, docker-compose brings up infra, CI green |
| M2 | Storage layer | Migrations + repos with integration tests |
| M3 | Queue layer | NATS publisher/subscriber with JetStream + DLQ |
| M4 | LLM provider | OpenAI client with retry, budget guard, Redis cache wrapper |
| M5 | session-reader | MTProto reads pool, publishes RawPost, persists last_seen_msg_id |
| M6 | Normalizer | Subscribes to `ingest.raw`, cleans, lang-detects, persists posts, publishes NormalizedPost |
| M7 | Dedup pipeline | MinHash → embeddings → LLM-merge, posts get cluster_id |
| M8 | Classifier + Ranker | Topics, severity, headline, summary, score |
| M9 | Digest assembler + Bot sender | Per-user filter, formatting, send, anti-repeat record |
| M10 | Scheduler | Cron triggers digest assembly with advisory lock |
| M11 | Bot API commands | `/start`, `/digest`, `/settings`, `/topics`, `/threshold` |
| M12 | Observability | Prometheus + OTel + structured logs end-to-end |
| M13 | Golden dataset + dogfooding | Eval harness, manual QA loop |

Each milestone ends in a tagged commit. Milestones build on each other strictly in order.

```go
// internal/config/config.go
package config

import (
	"errors"
	"fmt"
	"os"
	"strings"

	"gopkg.in/yaml.v3"
)

type Config struct {
	PostgresDSN  string  `yaml:"postgres_dsn"`
	RedisAddr    string  `yaml:"redis_addr"`
	NATSURL      string  `yaml:"nats_url"`
	BotToken     string  `yaml:"bot_token"`
	OpenAIAPIKey string  `yaml:"openai_api_key"`
	ChannelsFile string  `yaml:"channels_file"`
	MTProto      MTProto `yaml:"mtproto"`
	LLM          LLM     `yaml:"llm"`
}

type MTProto struct {
	APIID       int    `yaml:"api_id"`
	APIHash     string `yaml:"api_hash"`
	SessionPath string `yaml:"session_path"`
}

type LLM struct {
	MonthlyBudgetUSD float64 `yaml:"monthly_budget_usd"`
	EmbeddingModel   string  `yaml:"embedding_model"`
	ClassifierModel  string  `yaml:"classifier_model"`
}

func Load(path string) (*Config, error) {
	raw, err := os.ReadFile(path)
	if err != nil {
		return nil, fmt.Errorf("read config: %w", err)
	}
	var cfg Config
	if err := yaml.Unmarshal(raw, &cfg); err != nil {
		return nil, fmt.Errorf("parse yaml: %w", err)
	}
	applyEnvOverrides(&cfg)
	if err := cfg.validate(); err != nil {
		return nil, err
	}
	return &cfg, nil
}

func applyEnvOverrides(c *Config) {
	overrides := map[string]*string{
		"MUTE_POSTGRES_DSN":   &c.PostgresDSN,
		"MUTE_REDIS_ADDR":     &c.RedisAddr,
		"MUTE_NATS_URL":       &c.NATSURL,
		"MUTE_BOT_TOKEN":      &c.BotToken,
		"MUTE_OPENAI_API_KEY": &c.OpenAIAPIKey,
	}
	for k, p := range overrides {
		if v, ok := os.LookupEnv(k); ok && strings.TrimSpace(v) != "" {
			*p = v
		}
	}
}

func (c *Config) validate() error {
	var errs []error
	if c.PostgresDSN == "" {
		errs = append(errs, errors.New("postgres_dsn is required"))
	}
	if c.RedisAddr == "" {
		errs = append(errs, errors.New("redis_addr is required"))
	}
	if c.NATSURL == "" {
		errs = append(errs, errors.New("nats_url is required"))
	}
	if c.LLM.EmbeddingModel == "" {
		c.LLM.EmbeddingModel = "text-embedding-3-small"
	}
	if c.LLM.ClassifierModel == "" {
		c.LLM.ClassifierModel = "gpt-4o-mini"
	}
	if c.LLM.MonthlyBudgetUSD == 0 {
		c.LLM.MonthlyBudgetUSD = 50
	}
	return errors.Join(errs...)
}
```

```sql
-- migrations/0001_initial.up.sql
CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE channels (
  id              bigserial PRIMARY KEY,
  tg_channel_id   bigint UNIQUE NOT NULL,
  username        text,
  title           text,
  authority_score int NOT NULL DEFAULT 50,
  added_by        bigint,
  active          bool NOT NULL DEFAULT true,
  created_at      timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE clusters (
  id              bigserial PRIMARY KEY,
  headline        text NOT NULL DEFAULT '',
  summary         text NOT NULL DEFAULT '',
  topics          text[] NOT NULL DEFAULT '{}',
  severity        int NOT NULL DEFAULT 0,
  coverage        int NOT NULL DEFAULT 1,
  score           real NOT NULL DEFAULT 0,
  first_seen_at   timestamptz NOT NULL DEFAULT now(),
  last_updated_at timestamptz NOT NULL DEFAULT now(),
  status          text NOT NULL DEFAULT 'active'
);
CREATE INDEX clusters_last_updated_score_idx ON clusters (last_updated_at DESC, score DESC);
CREATE INDEX clusters_topics_gin_idx        ON clusters USING gin(topics);

CREATE TABLE posts (
  id           bigserial PRIMARY KEY,
  channel_id   bigint NOT NULL REFERENCES channels(id),
  tg_msg_id    bigint NOT NULL,
  text_raw     text NOT NULL,
  text_clean   text NOT NULL,
  text_hash    bytea NOT NULL,
  lang         text,
  posted_at    timestamptz NOT NULL,
  ingested_at  timestamptz NOT NULL DEFAULT now(),
  cluster_id   bigint REFERENCES clusters(id),
  UNIQUE(channel_id, tg_msg_id)
```

```go
// internal/storage/postgres/pool.go
package postgres

import (
	"context"
	"fmt"

	"github.com/jackc/pgx/v5/pgxpool"
	pgxvec "github.com/pgvector/pgvector-go/pgx"
)

type Pool struct {
	pool *pgxpool.Pool
}

func NewPool(ctx context.Context, dsn string) (*Pool, error) {
	cfg, err := pgxpool.ParseConfig(dsn)
	if err != nil {
		return nil, fmt.Errorf("parse dsn: %w", err)
	}
	cfg.AfterConnect = func(ctx context.Context, conn *pgxpool.Conn) error {
		return pgxvec.RegisterTypes(ctx, conn.Conn())
	}
	p, err := pgxpool.NewWithConfig(ctx, cfg)
	if err != nil {
		return nil, fmt.Errorf("new pool: %w", err)
	}
	if err := p.Ping(ctx); err != nil {
		p.Close()
		return nil, fmt.Errorf("ping: %w", err)
	}
	return &Pool{pool: p}, nil
}

func (p *Pool) Pool() *pgxpool.Pool { return p.pool }
func (p *Pool) Close()              { p.pool.Close() }
```

```go
func NewChannelsRepo(p *Pool) *ChannelsRepo { return &ChannelsRepo{p: p} }

func (r *ChannelsRepo) Upsert(ctx context.Context, in ChannelInsert) (int64, error) {
	const q = `
		INSERT INTO channels (tg_channel_id, username, title, authority_score)
		VALUES ($1,$2,$3,$4)
		ON CONFLICT (tg_channel_id) DO UPDATE
		   SET username = EXCLUDED.username,
		       title = EXCLUDED.title,
		       authority_score = EXCLUDED.authority_score
		RETURNING id`
	var id int64
	if err := r.p.Pool().QueryRow(ctx, q, in.TGChannelID, in.Username, in.Title, in.Authority).Scan(&id); err != nil {
		return 0, fmt.Errorf("upsert channel: %w", err)
	}
	return id, nil
}

		Scan(&c.ID, &c.TGChannelID, &c.Username, &c.Title, &c.Authority, &c.Active); err != nil {
```

```go
// internal/storage/postgres/clusters_repo.go
package postgres

import (
	"context"
	"fmt"
	"time"
)

type Cluster struct {
	ID            int64
	Headline      string
	Summary       string
	Topics        []string
	Severity      int
	Coverage      int
	Score         float32
	FirstSeenAt   time.Time
	LastUpdatedAt time.Time
	Status        string
}

type ClusterMeta struct {
	Headline string
	Summary  string
	Topics   []string
	Severity int
}

type ClustersRepo struct{ p *Pool }

func NewClustersRepo(p *Pool) *ClustersRepo { return &ClustersRepo{p: p} }

func (r *ClustersRepo) Create(ctx context.Context) (int64, error) {
	var id int64
	if err := r.p.Pool().QueryRow(ctx,
		`INSERT INTO clusters DEFAULT VALUES RETURNING id`).Scan(&id); err != nil {
		return 0, fmt.Errorf("create cluster: %w", err)
	}
	return id, nil
}

func (r *ClustersRepo) UpdateMeta(ctx context.Context, id int64, m ClusterMeta) error {
	_, err := r.p.Pool().Exec(ctx,
		`UPDATE clusters SET headline=$2, summary=$3, topics=$4, severity=$5, last_updated_at=now()
		 WHERE id=$1`, id, m.Headline, m.Summary, m.Topics, m.Severity)
	if err != nil { return fmt.Errorf("update meta: %w", err) }
	return nil
}

func (r *ClustersRepo) SetScore(ctx context.Context, id int64, score float32) error {
	_, err := r.p.Pool().Exec(ctx,
		`UPDATE clusters SET score=$2, last_updated_at=now() WHERE id=$1`, id, score)
	return err
}

func (r *ClustersRepo) IncrementCoverage(ctx context.Context, id int64) error {
	_, err := r.p.Pool().Exec(ctx,
		`UPDATE clusters SET coverage = coverage + 1, last_updated_at=now() WHERE id=$1`, id)
	return err
}

func (r *ClustersRepo) Get(ctx context.Context, id int64) (Cluster, error) {
	const q = `SELECT id, headline, summary, topics, severity, coverage, score,
	                  first_seen_at, last_updated_at, status
	           FROM clusters WHERE id = $1`
	var c Cluster
	if err := r.p.Pool().QueryRow(ctx, q, id).Scan(
		&c.ID, &c.Headline, &c.Summary, &c.Topics, &c.Severity, &c.Coverage, &c.Score,
		&c.FirstSeenAt, &c.LastUpdatedAt, &c.Status,
	); err != nil { return Cluster{}, fmt.Errorf("get cluster: %w", err) }
	return c, nil
}

type ClusterFilter struct {
	Topics     []string
	MinScore   float32
	SinceTime  time.Time
	ExcludeIDs []int64
	Limit      int
}

func (r *ClustersRepo) Search(ctx context.Context, f ClusterFilter) ([]Cluster, error) {
	const q = `SELECT id, headline, summary, topics, severity, coverage, score,
	                  first_seen_at, last_updated_at, status
	           FROM clusters
	           WHERE status='active'
	             AND last_updated_at >= $1
	             AND score >= $2
	             AND topics && $3::text[]
	             AND NOT (id = ANY($4::bigint[]))
	           ORDER BY score DESC
	           LIMIT $5`
	rows, err := r.p.Pool().Query(ctx, q, f.SinceTime, f.MinScore, f.Topics, f.ExcludeIDs, f.Limit)
	if err != nil { return nil, fmt.Errorf("search: %w", err) }
	defer rows.Close()
	out := make([]Cluster, 0, f.Limit)
	for rows.Next() {
		var c Cluster
		if err := rows.Scan(&c.ID, &c.Headline, &c.Summary, &c.Topics, &c.Severity, &c.Coverage,
			&c.Score, &c.FirstSeenAt, &c.LastUpdatedAt, &c.Status); err != nil { return nil, err }
		out = append(out, c)
	}
```

```go
	PostID   int64
	Distance float32

type NearestParams struct {
	Limit             int
	MaxCosineDistance float32
	SinceWindowSecs   int
}

func (r *EmbeddingsRepo) NearestNeighbors(ctx context.Context, v pgvector.Vector, p NearestParams) ([]Neighbor, error) {
	if p.SinceWindowSecs == 0 { p.SinceWindowSecs = 48 * 3600 }
	const q = `
		SELECT e.post_id, (e.embedding <=> $1) AS dist
		FROM post_embeddings e
		JOIN posts po ON po.id = e.post_id
		WHERE po.ingested_at >= now() - ($2 || ' seconds')::interval
		  AND (e.embedding <=> $1) <= $3
		ORDER BY e.embedding <=> $1
		LIMIT $4`
	rows, err := r.p.Pool().Query(ctx, q, v, p.SinceWindowSecs, p.MaxCosineDistance, p.Limit)
	if err != nil { return nil, fmt.Errorf("nn query: %w", err) }
	defer rows.Close()
	out := make([]Neighbor, 0, p.Limit)
	for rows.Next() {
		var n Neighbor
		if err := rows.Scan(&n.PostID, &n.Distance); err != nil { return nil, err }
```

```go
import (
	"context"
```

```go
	"github.com/nats-io/nats.go"

type Conn struct {
	nc *nats.Conn
	js jetstream.JetStream
}

func Connect(ctx context.Context, url string) (*Conn, error) {
	nc, err := nats.Connect(url, nats.MaxReconnects(-1))
	if err != nil { return nil, fmt.Errorf("nats connect: %w", err) }
	js, err := jetstream.New(nc)
	if err != nil { nc.Close(); return nil, fmt.Errorf("jetstream new: %w", err) }
	return &Conn{nc: nc, js: js}, nil
}

func (c *Conn) Close() { c.nc.Close() }
func (c *Conn) JS() jetstream.JetStream { return c.js }
func (c *Conn) NC() *nats.Conn          { return c.nc }

func (c *Conn) EnsureStreams(ctx context.Context) error {
	configs := []jetstream.StreamConfig{
		{
			Name:      StreamIngest,
			Subjects:  []string{SubjectRaw, SubjectNormalized, "ingest.>.dlq"},
			Retention: jetstream.WorkQueuePolicy,
			MaxAge:    72 * 3600 * 1_000_000_000,
		},
		{
			Name:      StreamClusters,
			Subjects:  []string{SubjectClusterUpdate, SubjectClusterScored, "cluster.>.dlq"},
			Retention: jetstream.WorkQueuePolicy,
			MaxAge:    72 * 3600 * 1_000_000_000,
		},
		{
			Name:      StreamDelivery,
			Subjects:  []string{SubjectDeliverySched, SubjectDeliveryPro, SubjectDeliveryFree, "delivery.>.dlq"},
			Retention: jetstream.WorkQueuePolicy,
			MaxAge:    72 * 3600 * 1_000_000_000,
		},
	}

		_, err := c.js.CreateStream(ctx, cfg)

			_, uerr := c.js.UpdateStream(ctx, cfg)

		}
	}
```

```go
// internal/queue/subscriber.go
package queue

import (
	"context"
	"fmt"
	"log/slog"
	"time"

	"github.com/nats-io/nats.go/jetstream"
)

type Handler func(ctx context.Context, data []byte) error

type SubscribeConfig struct {
	Stream     string
	Subject    string
	Durable    string
	MaxDeliver int
	Handler    Handler
}

type Subscriber struct {
	c *Conn
}

func NewSubscriber(c *Conn) *Subscriber { return &Subscriber{c: c} }

func (s *Subscriber) Run(ctx context.Context, cfg SubscribeConfig) error {
	if cfg.MaxDeliver <= 0 { cfg.MaxDeliver = 5 }
	cons, err := s.c.JS().CreateOrUpdateConsumer(ctx, cfg.Stream, jetstream.ConsumerConfig{
		Durable:        cfg.Durable,
		FilterSubject:  cfg.Subject,
		AckPolicy:      jetstream.AckExplicitPolicy,
		MaxDeliver:     cfg.MaxDeliver,
		AckWait:        30 * time.Second,
		MaxAckPending:  256,
	})
	if err != nil { return fmt.Errorf("create consumer: %w", err) }

	iter, err := cons.Messages()
	if err != nil { return fmt.Errorf("messages: %w", err) }
	defer iter.Stop()

	pub := NewPublisher(s.c)

	for {
		select {
		case <-ctx.Done():
			return ctx.Err()
		default:
		}
		msg, err := iter.Next()
		if err != nil {
			if ctx.Err() != nil { return ctx.Err() }
			slog.Warn("iter.Next", "err", err)
			time.Sleep(time.Second)
			continue
		}
		md, _ := msg.Metadata()
		if err := cfg.Handler(ctx, msg.Data()); err != nil {
			if md != nil && md.NumDelivered >= uint64(cfg.MaxDeliver) {
				slog.Error("DLQ", "subject", cfg.Subject, "err", err, "delivered", md.NumDelivered)
				_ = pub.Publish(ctx, cfg.Subject+DLQSuffix, map[string]any{
					"original": string(msg.Data()),
					"error":    err.Error(),
				})
				_ = msg.Ack()
				continue
			}
			_ = msg.NakWithDelay(backoff(int(md.NumDelivered)))
			continue
		}
		_ = msg.Ack()
	}
}

func backoff(n int) time.Duration {
	switch {
	case n <= 1: return 2 * time.Second
	case n == 2: return 5 * time.Second
	case n == 3: return 15 * time.Second
	default:     return 60 * time.Second
	}
}
```

```go
// internal/llm/budget.go
package llm

import (
	"context"
	"errors"
	"sync"
	"time"
)

type BudgetState int

const (
	BudgetOK BudgetState = iota
	BudgetWarn        // >= 80%
	BudgetDegraded    // >= 90%
	BudgetBlocked     // >= 100%
)

var ErrBudgetExceeded = errors.New("monthly LLM budget exceeded")

type BudgetConfig struct {
	MonthlyUSD float64
}

type BudgetGuard struct {
	mu         sync.Mutex
	cfg        BudgetConfig
	month      time.Month
	year       int
	spentUSD   float64
	now        func() time.Time
}

func NewBudgetGuard(cfg BudgetConfig) *BudgetGuard {
	return &BudgetGuard{cfg: cfg, now: time.Now}
}

func (b *BudgetGuard) Charge(ctx context.Context, costUSD float64) error {
	b.mu.Lock()
	defer b.mu.Unlock()
	b.rollIfNeededLocked()
	if b.spentUSD+costUSD > b.cfg.MonthlyUSD*1.10 {
		return ErrBudgetExceeded
	}
	b.spentUSD += costUSD
	return nil
}

func (b *BudgetGuard) State() BudgetState {
	b.mu.Lock()
	defer b.mu.Unlock()
	b.rollIfNeededLocked()
	ratio := b.spentUSD / b.cfg.MonthlyUSD
	switch {
	case ratio >= 1.0:  return BudgetBlocked
	case ratio >= 0.90: return BudgetDegraded
	case ratio >= 0.80: return BudgetWarn
	default:            return BudgetOK
	}
}

func (b *BudgetGuard) SpentUSD() float64 {
	b.mu.Lock(); defer b.mu.Unlock()
	b.rollIfNeededLocked()
	return b.spentUSD
}

func (b *BudgetGuard) rollIfNeededLocked() {
	t := b.now().UTC()
	if t.Year() != b.year || t.Month() != b.month {
		b.year, b.month = t.Year(), t.Month()
		b.spentUSD = 0
	}
}
```

```go
type Publisher interface {
	Publish(ctx context.Context, subject string, payload any) error
}

type Reader struct {
	client *telegram.Client
	pub    Publisher
	subj   string
}

func NewReader(c *telegram.Client, pub Publisher) *Reader {
	return &Reader{client: c, pub: pub, subj: queue.SubjectRaw}

func ExtractRawPost(msg tg.MessageClass) (domain.RawPost, bool) {
	m, ok := msg.(*tg.Message)
	if !ok || m.Message == "" { return domain.RawPost{}, false }
	var chID int64
	switch p := m.PeerID.(type) {
	case *tg.PeerChannel: chID = p.ChannelID
	case *tg.PeerChat:    chID = p.ChatID
	default:              return domain.RawPost{}, false
	}
	return domain.RawPost{
		ChannelID: chID,
		TGMsgID:   int64(m.ID),
		Text:      m.Message,
		PostedAt:  time.Unix(int64(m.Date), 0).UTC(),
	}, true
}

func (r *Reader) Run(ctx context.Context) error {
	dispatcher := tg.NewUpdateDispatcher()
	dispatcher.OnNewChannelMessage(func(ctx context.Context, e tg.Entities, u *tg.UpdateNewChannelMessage) error {
		rp, ok := ExtractRawPost(u.Message)
		if !ok { return nil }
		if err := r.pub.Publish(ctx, r.subj, rp); err != nil {
			slog.Error("publish raw post", "err", err, "channel", rp.ChannelID, "msg", rp.TGMsgID)
		}
		return nil
	})

	mgr := updates.New(updates.Config{Handler: dispatcher})

	return r.client.Run(ctx, func(ctx context.Context) error {
		if err := mgr.Auth(ctx, r.client.API(), 0, false, false); err != nil {
			return fmt.Errorf("updates auth: %w", err)
		}
		<-ctx.Done()
		return ctx.Err()
	})
```

```go
// internal/mtproto/catchup.go
package mtproto

import (
	"context"
	"fmt"

	"github.com/gotd/td/tg"

	"github.com/virsi/mute-bot/internal/domain"
	"github.com/virsi/mute-bot/internal/queue"
)

type StateRepo interface {
	GetLastMsgID(ctx context.Context, channelID int64) (int64, error)
	UpsertLastMsgID(ctx context.Context, channelID, lastMsgID int64) error
}

type Catchup struct {
	api   *tg.Client
	state StateRepo
	pub   Publisher
	limit int
}

func NewCatchup(api *tg.Client, state StateRepo, pub Publisher) *Catchup {
	return &Catchup{api: api, state: state, pub: pub, limit: 100}
}

func (c *Catchup) Backfill(ctx context.Context, ch ChannelRef, localChannelDBID int64) error {
	last, err := c.state.GetLastMsgID(ctx, localChannelDBID)
	if err != nil { return err }
	hist, err := c.api.MessagesGetHistory(ctx, &tg.MessagesGetHistoryRequest{
		Peer:     &tg.InputPeerChannel{ChannelID: ch.ChannelID, AccessHash: ch.AccessHash},
		OffsetID: 0, Limit: c.limit, MinID: int(last),
	})
	if err != nil { return fmt.Errorf("get history: %w", err) }
	msgs, ok := hist.AsModified()
	if !ok { return nil }
	var maxID int64
	for _, m := range msgs.GetMessages() {
		rp, ok := ExtractRawPost(m)
		if !ok { continue }
		if rp.TGMsgID <= last { continue }
		if err := c.pub.Publish(ctx, queue.SubjectRaw, domain.RawPost{
			ChannelID: ch.ChannelID, TGMsgID: rp.TGMsgID, Text: rp.Text, PostedAt: rp.PostedAt,
		}); err != nil {
			return fmt.Errorf("publish backfill: %w", err)
		}
		if rp.TGMsgID > maxID { maxID = rp.TGMsgID }
	}
	if maxID > 0 {
		if err := c.state.UpsertLastMsgID(ctx, localChannelDBID, maxID); err != nil {
			return err
		}
	}
	return nil
}
```

```go
package normalize

import (
	"regexp"
	"strings"
)

var (
	reURL      = regexp.MustCompile(`https?://\S+`)
	reMention  = regexp.MustCompile(`@\w+`)
	reHashtag  = regexp.MustCompile(`#\w+`)
	reSpaces   = regexp.MustCompile(`\s+`)
	reAdvert   = regexp.MustCompile(`(?im)\n+(\s*(реклама|advert|спонсор|promo|sponsored)[^\n]*)`)
)

func Clean(s string) string {
	s = reAdvert.ReplaceAllString(s, "")
	s = stripEmoji(s)
	s = reURL.ReplaceAllString(s, "")
	s = reMention.ReplaceAllString(s, "")
	s = reHashtag.ReplaceAllString(s, "")
	s = reSpaces.ReplaceAllString(s, " ")
	return strings.TrimSpace(s)
}

func stripEmoji(s string) string {
	var b strings.Builder
	for _, r := range s {
		if isEmoji(r) { continue }
		b.WriteRune(r)
	}
	return b.String()
}

func isEmoji(r rune) bool {
	switch {
	case r >= 0x1F600 && r <= 0x1F64F:
		return true
```

```go
// internal/dedup/minhash.go
package dedup

import (
	"hash/fnv"
	"math"
	"strings"
)

type MinHashConfig struct {
	NumHashes   int
	ShingleSize int
}

type MinHash struct {
	cfg MinHashConfig
}

type Signature []uint32

func NewMinHash(cfg MinHashConfig) *MinHash {
	if cfg.NumHashes <= 0 { cfg.NumHashes = 128 }
	if cfg.ShingleSize <= 0 { cfg.ShingleSize = 5 }
	return &MinHash{cfg: cfg}
}

func (m *MinHash) Sign(text string) Signature {
	tokens := tokenize(strings.ToLower(text))
	shingles := shingleWords(tokens, m.cfg.ShingleSize)
	sig := make(Signature, m.cfg.NumHashes)
	for i := range sig { sig[i] = math.MaxUint32 }
	for s := range shingles {
		base := fnv32(s)
		for i := 0; i < m.cfg.NumHashes; i++ {
			h := base*uint32(2654435761) + uint32(i)*uint32(40503)
			if h < sig[i] { sig[i] = h }
		}
	}
	return sig
}

func Jaccard(a, b Signature) float64 {
	if len(a) != len(b) || len(a) == 0 { return 0 }
	eq := 0
	for i := range a {
		if a[i] == b[i] { eq++ }
	}
	return float64(eq) / float64(len(a))
}

func tokenize(s string) []string {
	var out []string
	var b strings.Builder
	for _, r := range s {
		switch {
		case r == ' ' || r == '\n' || r == '\t' || r == '.' || r == ',' || r == ':' || r == ';' || r == '!' || r == '?':
			if b.Len() > 0 { out = append(out, b.String()); b.Reset() }
		default:
			b.WriteRune(r)
		}
	}
	if b.Len() > 0 { out = append(out, b.String()) }
	return out
}

func shingleWords(tokens []string, k int) map[string]struct{} {
	out := make(map[string]struct{}, len(tokens))
	if len(tokens) < k {
		out[strings.Join(tokens, " ")] = struct{}{}
		return out
	}
	for i := 0; i+k <= len(tokens); i++ {
		out[strings.Join(tokens[i:i+k], " ")] = struct{}{}
	}
	return out
}

func fnv32(s string) uint32 {
	h := fnv.New32a()
	_, _ = h.Write([]byte(s))
	return h.Sum32()
}
```

```go
package redis

import (
	"context"
	"crypto/sha1"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"time"
)

type MinHashIndexConfig struct {
	Bands       int
	RowsPerBand int
	TTLSecs     int
}

type MinHashIndex struct {
	c   *Client
	cfg MinHashIndexConfig
}

func NewMinHashIndex(c *Client, cfg MinHashIndexConfig) *MinHashIndex {
	if cfg.Bands == 0 { cfg.Bands = 16 }
	if cfg.RowsPerBand == 0 { cfg.RowsPerBand = 8 }
	if cfg.TTLSecs == 0 { cfg.TTLSecs = 48 * 3600 }
	return &MinHashIndex{c: c, cfg: cfg}
}

func (m *MinHashIndex) Add(ctx context.Context, postID int64, sig []uint32) error {
	if len(sig) != m.cfg.Bands*m.cfg.RowsPerBand {
		return fmt.Errorf("signature size %d != bands*rows %d", len(sig), m.cfg.Bands*m.cfg.RowsPerBand)
	}
	pipe := m.c.RDB().Pipeline()
	for b := 0; b < m.cfg.Bands; b++ {
		key := m.bandKey(b, sig[b*m.cfg.RowsPerBand:(b+1)*m.cfg.RowsPerBand])
		pipe.SAdd(ctx, key, postID)
		pipe.Expire(ctx, key, time.Duration(m.cfg.TTLSecs)*time.Second)
	}
	_, err := pipe.Exec(ctx)
	if err != nil { return fmt.Errorf("pipeline: %w", err) }
	return nil
}

func (m *MinHashIndex) Candidates(ctx context.Context, sig []uint32, limit int) ([]int64, error) {
	if len(sig) != m.cfg.Bands*m.cfg.RowsPerBand {
		return nil, fmt.Errorf("signature size mismatch")
	}
	seen := make(map[int64]struct{}, limit)
	for b := 0; b < m.cfg.Bands; b++ {
		key := m.bandKey(b, sig[b*m.cfg.RowsPerBand:(b+1)*m.cfg.RowsPerBand])
		ids, err := m.c.RDB().SMembers(ctx, key).Result()
		if err != nil { return nil, fmt.Errorf("smembers: %w", err) }
		for _, idStr := range ids {
			var id int64
			fmt.Sscanf(idStr, "%d", &id)
			seen[id] = struct{}{}
		}
	}
	out := make([]int64, 0, len(seen))
	for id := range seen { out = append(out, id) }
	if len(out) > limit { out = out[:limit] }
	return out, nil
}

func (m *MinHashIndex) bandKey(band int, rows []uint32) string {
	raw, _ := json.Marshal(rows)
	sum := sha1.Sum(raw)
	return fmt.Sprintf("mh:b%d:%s", band, hex.EncodeToString(sum[:]))
}
```

```go
func NewEmbeddingCache(c *Client, ttlSecs int) *EmbeddingCache {
	if ttlSecs == 0 { ttlSecs = 7 * 24 * 3600 }
	return &EmbeddingCache{c: c, ttlSecs: ttlSecs}
}

func (e *EmbeddingCache) Get(ctx context.Context, hash [32]byte) ([]float32, bool, error) {
	key := "emb:" + hex.EncodeToString(hash[:])
	raw, err := e.c.RDB().Get(ctx, key).Bytes()
	if err != nil {
		if err.Error() == "redis: nil" { return nil, false, nil }
		return nil, false, fmt.Errorf("get: %w", err)
	}
	var v []float32
	if err := json.Unmarshal(raw, &v); err != nil { return nil, false, fmt.Errorf("decode: %w", err) }
	return v, true, nil
}

func (e *EmbeddingCache) Set(ctx context.Context, hash [32]byte, v []float32) error {
	raw, _ := json.Marshal(v)
	key := "emb:" + hex.EncodeToString(hash[:])
	return e.c.RDB().Set(ctx, key, raw, time.Duration(e.ttlSecs)*time.Second).Err()
}
```

```go
	"context"
	"fmt"
	"time"

	Add(ctx context.Context, postID int64, sig []uint32) error
	Candidates(ctx context.Context, sig []uint32, limit int) ([]int64, error)

type EmbedderIface interface {
	EmbedOne(ctx context.Context, text string, hash [32]byte) ([]float32, error)

type EmbeddingsStore interface {
	Store(ctx context.Context, postID int64, v pgvector.Vector, model string) error
	NearestNeighbors(ctx context.Context, v pgvector.Vector, p postgres.NearestParams) ([]postgres.Neighbor, error)

	Create(ctx context.Context) (int64, error)
	IncrementCoverage(ctx context.Context, id int64) error
}

type PostsStore interface {
	GetClusterID(ctx context.Context, postID int64) (int64, error)
	AttachCluster(ctx context.Context, postID, clusterID int64) error
}

type Publisher interface {
	Publish(ctx context.Context, subject string, payload any) error
}

type MatcherDeps struct {
	MinHash      *MinHash
	MinHashIndex MHIndex
	Embedder     EmbedderIface
	Embeddings   EmbeddingsStore
	Clusters     ClustersStore
	Posts        PostsStore
	Publisher    Publisher
	Model        string
	NearLimit    int
	MaxDistance  float32 // cosine distance threshold; 0.15 ≈ similarity 0.85
}

type Matcher struct {
	d MatcherDeps
}

func NewMatcher(d MatcherDeps) *Matcher {
	if d.MinHash == nil   { d.MinHash = NewMinHash(MinHashConfig{NumHashes: 128, ShingleSize: 5}) }
	if d.NearLimit == 0   { d.NearLimit = 5 }
	if d.MaxDistance == 0 { d.MaxDistance = 0.15 }
	if d.Model == ""      { d.Model = "text-embedding-3-small" }
	return &Matcher{d: d}
}

type MatchInput struct {
	PostID    int64
	TextClean string
	Hash      [32]byte
}

func (m *Matcher) Match(ctx context.Context, in MatchInput) error {
	sig := m.d.MinHash.Sign(in.TextClean)

	candidates, err := m.d.MinHashIndex.Candidates(ctx, sig, 16)
	if err != nil { return fmt.Errorf("mh candidates: %w", err) }

	for _, cid := range candidates {
		clusterID, err := m.d.Posts.GetClusterID(ctx, cid)
		if err != nil { continue }
		if clusterID == 0 { continue }
		if err := m.d.Posts.AttachCluster(ctx, in.PostID, clusterID); err != nil { return err }
		if err := m.d.Clusters.IncrementCoverage(ctx, clusterID); err != nil { return err }
		_ = m.d.MinHashIndex.Add(ctx, in.PostID, sig)
		return m.d.Publisher.Publish(ctx, queue.SubjectClusterUpdate, map[string]any{
			"cluster_id": clusterID, "via": "minhash", "ts": time.Now().UTC(),
		})
	}

	vec, err := m.d.Embedder.EmbedOne(ctx, in.TextClean, in.Hash)
	if err != nil { return fmt.Errorf("embed: %w", err) }
	pv := pgvector.NewVector(vec)
	if err := m.d.Embeddings.Store(ctx, in.PostID, pv, m.d.Model); err != nil { return err }

	near, err := m.d.Embeddings.NearestNeighbors(ctx, pv, postgres.NearestParams{
		Limit: m.d.NearLimit, MaxCosineDistance: m.d.MaxDistance,
	})
	if err != nil { return fmt.Errorf("nn: %w", err) }

	for _, n := range near {
		if n.PostID == in.PostID { continue }
		clusterID, err := m.d.Posts.GetClusterID(ctx, n.PostID)
		if err != nil || clusterID == 0 { continue }
		if err := m.d.Posts.AttachCluster(ctx, in.PostID, clusterID); err != nil { return err }
		if err := m.d.Clusters.IncrementCoverage(ctx, clusterID); err != nil { return err }
		_ = m.d.MinHashIndex.Add(ctx, in.PostID, sig)
		return m.d.Publisher.Publish(ctx, queue.SubjectClusterUpdate, map[string]any{
			"cluster_id": clusterID, "via": "embedding", "ts": time.Now().UTC(),
		})
	}

	clusterID, err := m.d.Clusters.Create(ctx)
	if err != nil { return fmt.Errorf("create cluster: %w", err) }
	if err := m.d.Posts.AttachCluster(ctx, in.PostID, clusterID); err != nil { return err }
	_ = m.d.MinHashIndex.Add(ctx, in.PostID, sig)
	return m.d.Publisher.Publish(ctx, queue.SubjectClusterUpdate, map[string]any{
		"cluster_id": clusterID, "via": "new", "ts": time.Now().UTC(),
	})
}
```

```go
	user := fmt.Sprintf("Snippet A:\n%s\n\nSnippet B:\n%s", a, b)
```

```go
	body = strings.ReplaceAll(body, "{{LANG}}", lang)
	body = strings.ReplaceAll(body, "{{POSTS}}", joinPosts(posts))

	resp, err := c.d.LLM.Chat(ctx, llm.ChatRequest{

	})

		return Result{}, nil // graceful fallback — caller treats empty Topics as "Other"

	if r.Severity < 0   { r.Severity = 0   }
	if r.Severity > 100 { r.Severity = 100 }
	return r, nil
}

		out += fmt.Sprintf("[%d] %s\n", i+1, s)
```

```go
	Publish(ctx context.Context, subject string, payload any) error

	Classifier ClassifierIface
	Posts      PostsReader
	Clusters   ClustersUpdater
	Publisher  Publisher
	Debounce   time.Duration

type Worker struct {
	d       WorkerDeps
	mu      sync.Mutex
	pending map[int64]time.Time
}

func NewWorker(d WorkerDeps) *Worker {
	if d.Debounce == 0 { d.Debounce = 60 * time.Second }
	return &Worker{d: d, pending: map[int64]time.Time{}}

func (w *Worker) Handle(_ context.Context, data []byte) error {
	var evt struct{ ClusterID int64 `json:"cluster_id"` }
	if err := json.Unmarshal(data, &evt); err != nil { return fmt.Errorf("unmarshal: %w", err) }
	w.mu.Lock()
	w.pending[evt.ClusterID] = time.Now().Add(w.d.Debounce)
	w.mu.Unlock()
	return nil
}

func (w *Worker) runDebouncer(ctx context.Context) {
	t := time.NewTicker(w.d.Debounce / 2)
	defer t.Stop()
	for {
		select {
		case <-ctx.Done(): return
		case now := <-t.C:
			w.mu.Lock()
			ready := make([]int64, 0)
			for id, due := range w.pending {
				if !now.Before(due) {
					ready = append(ready, id)
					delete(w.pending, id)
				}
			}
			w.mu.Unlock()
			for _, id := range ready {
				if err := w.process(ctx, id); err != nil {
					// log only; will be re-pushed by next cluster.updated event
					continue
				}
			}
		}
	}
}

func (w *Worker) process(ctx context.Context, clusterID int64) error {
	texts, lang, err := w.d.Posts.ListTextsByCluster(ctx, clusterID)
	if err != nil { return fmt.Errorf("list texts: %w", err) }
	if len(texts) == 0 { return nil }
	res, err := w.d.Classifier.Classify(ctx, texts, lang)
	if err != nil { return fmt.Errorf("classify: %w", err) }
	meta := struct {
		Headline, Summary string
		Topics            []string
		Severity          int
	}{Headline: res.Headline, Summary: res.Summary, Topics: res.Topics, Severity: res.Severity}
	if err := w.d.Clusters.UpdateMeta(ctx, clusterID, meta); err != nil { return fmt.Errorf("update meta: %w", err) }
	return w.d.Publisher.Publish(ctx, queue.SubjectClusterScored, map[string]any{"cluster_id": clusterID})
}

func (w *Worker) Run(ctx context.Context) { w.runDebouncer(ctx) }
```

```go
	"github.com/virsi/mute-bot/internal/domain"

type Item struct {
	ClusterID int64
	Headline  string
	Summary   string
	Topics    []string
	Sources   []string
	Score     float32
}

type FormatOptions struct {
	Now   time.Time
	Title string
}

func Format(items []Item, opts FormatOptions) string {
	if len(items) == 0 { return "" }
	var b strings.Builder
	fmt.Fprintf(&b, "🗞 %s · %s\n\n", opts.Title, opts.Now.Format("02.01.2006"))

	byTopic := make(map[string][]Item)
	order := []string{}
	for _, it := range items {
		key := primaryTopic(it.Topics)
		if _, ok := byTopic[key]; !ok { order = append(order, key) }
		byTopic[key] = append(byTopic[key], it)
	}

	for _, topic := range order {
		fmt.Fprintf(&b, "📌 %s\n", topicTitle(topic))
		topicItems := byTopic[topic]
		sort.SliceStable(topicItems, func(i, j int) bool { return topicItems[i].Score > topicItems[j].Score })
		for i, it := range topicItems {
			fmt.Fprintf(&b, "%d. %s\n   %s\n   📡 %s\n\n", i+1, it.Headline, it.Summary, formatSources(it.Sources))
		}
	}
	return strings.TrimRight(b.String(), "\n")
}

func primaryTopic(topics []string) string {
	if len(topics) == 0 { return "other" }
	return topics[0]
}

func topicTitle(id string) string {
	for _, t := range domain.PresetTopics {
		if t.ID == id { return t.Title }
	}
	return "Прочее"
}

func formatSources(s []string) string {
	if len(s) == 0 { return "—" }
	out := make([]string, 0, len(s))
	for i, u := range s {
		if i >= 5 { break }
		out = append(out, "@"+u)
	}
	return strings.Join(out, ", ")
```

```go
// internal/bot/sender.go
package bot

import (
	"context"
	"fmt"
	"sync"
	"time"
)

type API interface {
	Send(ctx context.Context, chatID int64, text string) error
}

type SenderDeps struct {
	API           API
	PerChatPerSec int // tokens per second per chat
}

type Sender struct {
	d        SenderDeps
	mu       sync.Mutex
	buckets  map[int64]*bucket
}

type bucket struct {
	tokens     float64
	lastRefill time.Time
}

func NewSender(d SenderDeps) *Sender {
	if d.PerChatPerSec == 0 { d.PerChatPerSec = 1 }
	return &Sender{d: d, buckets: map[int64]*bucket{}}
}

func (s *Sender) SendDigest(ctx context.Context, chatID int64, text string) error {
	if err := s.wait(ctx, chatID); err != nil { return err }
	return s.d.API.Send(ctx, chatID, text)
}

func (s *Sender) wait(ctx context.Context, chatID int64) error {
	for {
		s.mu.Lock()
		b, ok := s.buckets[chatID]
		now := time.Now()
		if !ok {
			b = &bucket{tokens: float64(s.d.PerChatPerSec), lastRefill: now}
			s.buckets[chatID] = b
		}
		elapsed := now.Sub(b.lastRefill).Seconds()
		b.tokens += elapsed * float64(s.d.PerChatPerSec)
		if b.tokens > float64(s.d.PerChatPerSec) { b.tokens = float64(s.d.PerChatPerSec) }
		b.lastRefill = now
		if b.tokens >= 1 {
			b.tokens--
			s.mu.Unlock()
			return nil
		}
		need := (1 - b.tokens) / float64(s.d.PerChatPerSec)
		s.mu.Unlock()
		select {
```

```go
package scheduler

import (
	"context"
	"fmt"

	"github.com/virsi/mute-bot/internal/storage/postgres"
)

func TryLock(ctx context.Context, p *postgres.Pool, key int64) (bool, error) {
	var ok bool
	if err := p.Pool().QueryRow(ctx, `SELECT pg_try_advisory_lock($1)`, key).Scan(&ok); err != nil {
		return false, fmt.Errorf("advisory lock: %w", err)
	}
	return ok, nil
}

func Unlock(ctx context.Context, p *postgres.Pool, key int64) error {
	_, err := p.Pool().Exec(ctx, `SELECT pg_advisory_unlock($1)`, key)
	return err
}
```

```go
	"context"
```

```go
	"github.com/go-co-op/gocron/v2"

	"github.com/virsi/mute-bot/internal/queue"

type UserSchedule struct {
	UserID   int64
	TGUserID int64
	Times    []string // "HH:MM"
	TZ       string
}

type LoadUsersFunc func(ctx context.Context) ([]UserSchedule, error)

type Publisher interface {
	Publish(ctx context.Context, subject string, payload any) error
}

type CronDeps struct {
	LoadUsers LoadUsersFunc
	Publisher Publisher
	Reload    time.Duration
}

type Cron struct {
	s   gocron.Scheduler
	d   CronDeps
}

func NewCron(d CronDeps) (*Cron, error) {
	if d.Reload == 0 { d.Reload = 5 * time.Minute }
	s, err := gocron.NewScheduler()
	if err != nil { return nil, fmt.Errorf("new scheduler: %w", err) }
	return &Cron{s: s, d: d}, nil
}

func (c *Cron) Start(ctx context.Context) error {
	if err := c.reload(ctx); err != nil { return err }
	c.s.Start()
	go func() {
		t := time.NewTicker(c.d.Reload); defer t.Stop()
		for {
			select {
			case <-ctx.Done(): return
			case <-t.C: _ = c.reload(ctx)
			}
		}
	}()
	return nil
}

func (c *Cron) Stop() error { return c.s.Shutdown() }

func (c *Cron) reload(ctx context.Context) error {
	users, err := c.d.LoadUsers(ctx)
	if err != nil { return fmt.Errorf("load: %w", err) }
	for _, j := range c.s.Jobs() { _ = c.s.RemoveJob(j.ID()) }
	for _, u := range users {
		u := u
		loc, err := time.LoadLocation(u.TZ)
		if err != nil { loc = time.UTC }
		for _, hhmm := range u.Times {
			t, err := time.ParseInLocation("15:04", hhmm, loc)
			if err != nil { continue }
			_, err = c.s.NewJob(
				gocron.DailyJob(1, gocron.NewAtTimes(gocron.NewAtTime(uint(t.Hour()), uint(t.Minute()), 0))),
				gocron.NewTask(func() {
					_ = c.d.Publisher.Publish(context.Background(), queue.SubjectDeliverySched,
						map[string]any{"user_id": u.UserID, "tg_user_id": u.TGUserID, "channel": "digest"})
				}),
				gocron.WithLocation(loc),
			)
			if err != nil { continue }
		}
	}
	return nil
```

```go
package eval

import (
	"context"
	"os"
	"testing"

	"github.com/stretchr/testify/require"

	"github.com/virsi/mute-bot/internal/dedup"
)

func TestDedup_GoldenPrecisionRecall(t *testing.T) {
	if os.Getenv("EVAL") != "1" { t.Skip("set EVAL=1 to run") }
	recs, err := LoadCorpus("../fixtures/posts_corpus_v1.jsonl")
	require.NoError(t, err); require.NotEmpty(t, recs)

	mh := dedup.NewMinHash(dedup.MinHashConfig{NumHashes: 128, ShingleSize: 5})
	// pairs sharing cluster_label
	gold := pairsByCluster(recs)
	predicted := map[[2]string]bool{}
	for i := range recs {
		for j := i + 1; j < len(recs); j++ {
			s1 := mh.Sign(recs[i].Text); s2 := mh.Sign(recs[j].Text)
			if dedup.Jaccard(s1, s2) >= 0.5 {
				predicted[pair(recs[i].ID, recs[j].ID)] = true
			}
		}
	}
	prec, rec := precisionRecall(predicted, gold)
	t.Logf("precision=%.3f recall=%.3f", prec, rec)
	require.GreaterOrEqual(t, prec, 0.90)
	require.GreaterOrEqual(t, rec, 0.70)
}

func pair(a, b string) [2]string { if a < b { return [2]string{a,b} }; return [2]string{b,a} }
func pairsByCluster(recs []GoldRecord) map[[2]string]bool {
	by := map[string][]string{}
	for _, r := range recs { by[r.ClusterLabel] = append(by[r.ClusterLabel], r.ID) }
	out := map[[2]string]bool{}
	for _, ids := range by {
		for i := range ids { for j := i + 1; j < len(ids); j++ { out[pair(ids[i],ids[j])] = true } }
	}
	return out
}
func precisionRecall(pred, gold map[[2]string]bool) (float64, float64) {
	tp := 0; fp := 0
	for p := range pred { if gold[p] { tp++ } else { fp++ } }
	fn := 0
	for g := range gold { if !pred[g] { fn++ } }
	prec := float64(tp) / float64(tp+fp+1)
	rec  := float64(tp) / float64(tp+fn+1)
	return prec, rec
}
```
