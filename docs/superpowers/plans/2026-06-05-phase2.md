# Phase 2 Implementation Plan — News Digest Bot

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task.

**Goal:** Close all Phase 1 tech-debt, then ship multi-user Free/Pro tiering with TG Stars billing (99 XTR/month), Pro-only custom topics, and Pro-only real-time alerts. No trial. No ЮКасса. No weekly digest. No web cabinet.

**Architecture:** Keep the four existing processes (`tg-scraper`, `processor`, `bot-api`, `scheduler`). Phase 2 adds three internal subsystems inside the existing binaries:
- `internal/users/` (service over `users_repo` + `subscriptions_repo`) used by `bot-api` and `scheduler`.
- `internal/billing/` (PaymentProvider abstraction + Stars implementation + webhook handlers) used by `bot-api`.
- `internal/topics/` (custom topic embedding + cluster-match filter) used by `bot-api` (commands) and `processor` (digest filter).
- `internal/alerts/` (subscriber on `cluster.scored`, throttle, fan-out via existing `bot.Sender`) hosted in `cmd/processor`.

The borderline-reconciliation job (M1) is a new goroutine inside `cmd/processor` driven by a 5-min ticker.

**Tech Stack:** unchanged from Phase 1 — Go 1.23+, pgx/v5, slog, NATS JetStream, redis/go-redis/v9, gocron/v2, golang-migrate, testify, testcontainers-go, OpenAI client through `internal/llm` (ProxyAPI BaseURL already wired). New dep for envsubst: stdlib `regexp` only (no third-party).

**Spec references:**
- Phase 1 spec: `docs/superpowers/specs/2026-06-03-news-digest-bot-design.md`
- Phase 2 decisions delta: `docs/superpowers/specs/2026-06-05-phase2-decisions.md`
- Known issues backlog: `docs/known-issues.md`

func Load(path string) (*Config, error) {
    raw, err := os.ReadFile(path) // #nosec G304

var envRefRe = regexp.MustCompile(`\$\{([A-Z_][A-Z0-9_]*)\}`)

func expandEnv(in []byte) ([]byte, error) {
    var missing []string
    out := envRefRe.ReplaceAllFunc(in, func(match []byte) []byte {
        name := string(envRefRe.FindSubmatch(match)[1])
        v, ok := os.LookupEnv(name)
        if !ok {
            missing = append(missing, name)
            return match
        }
        return []byte(v)
    })
    if len(missing) > 0 {
        return nil, fmt.Errorf("env vars not set: %v", missing)
    }
    return out, nil
}

// internal/storage/redis/dedup_borderline.go
package redis

import (
    "context"
    "encoding/json"
    "fmt"

    "github.com/redis/go-redis/v9"
)

// BorderlinePair carries the minimum needed by the reconciler to call
// LLMJudge: the incoming post id, a candidate post id, and the cosine
// distance that put them into the borderline band.
type BorderlinePair struct {
    PostID      int64   `json:"post_id"`
    CandidateID int64   `json:"candidate_id"`
    Distance    float32 `json:"distance"`
}

// BorderlineQueue is the Redis list dedup:borderline drained by the
// reconciler goroutine.
type BorderlineQueue struct {
    rdb *redis.Client
    key string
    max int
}

// NewBorderlineQueue returns a queue bound to the dedup:borderline list.
// max caps the list size on every push so a stuck reconciler cannot make
// Redis OOM.
func NewBorderlineQueue(c *Client, max int) *BorderlineQueue {
    if max == 0 { max = 5000 }
    return &BorderlineQueue{rdb: c.RDB(), key: "dedup:borderline", max: max}
}

// Push appends a pair to the right of the list and trims the head if
// the list exceeds max.
func (q *BorderlineQueue) Push(ctx context.Context, p BorderlinePair) error {
    b, err := json.Marshal(p)
    if err != nil { return fmt.Errorf("marshal: %w", err) }
    pipe := q.rdb.TxPipeline()
    pipe.RPush(ctx, q.key, b)
    pipe.LTrim(ctx, q.key, int64(-q.max), -1)
    if _, err := pipe.Exec(ctx); err != nil { return fmt.Errorf("push: %w", err) }
    return nil
}

// Drain pops up to limit pairs from the left of the list and returns them.
func (q *BorderlineQueue) Drain(ctx context.Context, limit int) ([]BorderlinePair, error) {
    out := make([]BorderlinePair, 0, limit)
    for i := 0; i < limit; i++ {
        b, err := q.rdb.LPop(ctx, q.key).Bytes()
        if err == redis.Nil { break }
        if err != nil { return out, fmt.Errorf("lpop: %w", err) }
        var p BorderlinePair
        if err := json.Unmarshal(b, &p); err != nil { continue }
        out = append(out, p)
    }
    return out, nil
}

// internal/dedup/borderline_reconciler.go
package dedup

import (
    "context"
    "fmt"
    "log/slog"
    "time"

    redisstore "github.com/virsi/mute-bot/internal/storage/redis"
)

// BorderlineDrainer is the Redis list contract the reconciler needs.
type BorderlineDrainer interface {
    Drain(ctx context.Context, limit int) ([]redisstore.BorderlinePair, error)
}

// PostsForJudge is the storage surface the reconciler needs: load post text
// for two ids, and merge cluster ids.
type PostsForJudge interface {
    GetText(ctx context.Context, postID int64) (string, error)
    GetClusterID(ctx context.Context, postID int64) (int64, error)
    AttachCluster(ctx context.Context, postID, clusterID int64) error
}

// ClustersForJudge knows how to merge cluster B into cluster A (move all
// posts attached to B to A and mark B status='merged').
type ClustersForJudge interface {
    Merge(ctx context.Context, into, from int64) error
}

// JudgeDecider is the LLMJudge surface used here.
type JudgeDecider interface {
    Decide(ctx context.Context, a, b string) (bool, float64, error)
}

// ReconcilerDeps configures Reconciler.
type ReconcilerDeps struct {
    Queue    BorderlineDrainer
    Posts    PostsForJudge
    Clusters ClustersForJudge
    Judge    JudgeDecider
    Interval time.Duration // default 5 min
    BatchSize int          // default 50
    MinConfidence float64  // default 0.8
    Logger   *slog.Logger
}

// Reconciler drains the dedup:borderline list every Interval, asks LLMJudge
// per pair, and merges clusters when same=true && confidence >= MinConfidence.
type Reconciler struct{ d ReconcilerDeps }

func NewReconciler(d ReconcilerDeps) *Reconciler {
    if d.Interval == 0 { d.Interval = 5 * time.Minute }
    if d.BatchSize == 0 { d.BatchSize = 50 }
    if d.MinConfidence == 0 { d.MinConfidence = 0.8 }
    if d.Logger == nil { d.Logger = slog.Default() }
    return &Reconciler{d: d}
}

// Run blocks until ctx is cancelled, calling Step on every tick.
func (r *Reconciler) Run(ctx context.Context) error {
    t := time.NewTicker(r.d.Interval); defer t.Stop()
    for {
        select {
        case <-ctx.Done(): return ctx.Err()
        case <-t.C:
            if err := r.Step(ctx); err != nil {
                r.d.Logger.WarnContext(ctx, "borderline step", slog.Any("err", err))
            }
        }
    }
}

// Step drains BatchSize pairs and acts on each.
func (r *Reconciler) Step(ctx context.Context) error {
    pairs, err := r.d.Queue.Drain(ctx, r.d.BatchSize)
    if err != nil { return fmt.Errorf("drain: %w", err) }
    for _, p := range pairs {
        if err := r.judgeOne(ctx, p); err != nil {
            r.d.Logger.WarnContext(ctx, "judge pair",
                slog.Int64("post", p.PostID), slog.Int64("cand", p.CandidateID),
                slog.Any("err", err))
        }
    }
    return nil
}

func (r *Reconciler) judgeOne(ctx context.Context, p redisstore.BorderlinePair) error {
    a, err := r.d.Posts.GetText(ctx, p.PostID); if err != nil { return err }
    b, err := r.d.Posts.GetText(ctx, p.CandidateID); if err != nil { return err }
    same, conf, err := r.d.Judge.Decide(ctx, a, b); if err != nil { return err }
    if !same || conf < r.d.MinConfidence { return nil }
    cA, err := r.d.Posts.GetClusterID(ctx, p.PostID); if err != nil { return err }
    cB, err := r.d.Posts.GetClusterID(ctx, p.CandidateID); if err != nil { return err }
    if cA == cB || cA == 0 || cB == 0 { return nil }
    // Always merge the newer-id cluster into the older to keep canonical id stable.
    into, from := cA, cB
    if cB < cA { into, from = cB, cA }
    return r.d.Clusters.Merge(ctx, into, from)
}

// ListExpired returns ids of pro users whose tier_until is <= asOf.
func (r *UsersRepo) ListExpired(ctx context.Context, asOf time.Time) ([]int64, error) {
    const q = `SELECT id FROM users WHERE tier = 'pro' AND tier_until IS NOT NULL AND tier_until <= $1`
    rows, err := r.p.Pool().Query(ctx, q, asOf)
    if err != nil { return nil, fmt.Errorf("list expired: %w", err) }
    defer rows.Close()
    var out []int64
    for rows.Next() {
        var id int64
        if err := rows.Scan(&id); err != nil { return nil, err }
        out = append(out, id)
    }
    return out, rows.Err()
}

func (r *UsersRepo) SetTier(ctx context.Context, id int64, tier string, until *time.Time) error {
    if _, err := r.p.Pool().Exec(ctx,
        `UPDATE users SET tier = $2, tier_until = $3 WHERE id = $1`,
        id, tier, until); err != nil {
        return fmt.Errorf("set tier: %w", err)

func (s *Service) RegisterOnStart(ctx context.Context, tgUserID int64, username string) (postgres.User, bool, error) {
    u, created, err := s.users.GetOrCreate(ctx, tgUserID, username)
    if err != nil { return postgres.User{}, false, fmt.Errorf("get_or_create: %w", err) }

            AlertsEnabled: false, AlertThreshold: 85,
        }); err != nil {

        }
    }

// IsPro returns true when the user's tier is 'pro' AND tier_until is null
// or in the future.

    if u.Tier != "pro" { return false }
    if u.TierUntil == nil { return true }
    return u.TierUntil.After(s.now())
}

func (s *Service) GrantPro(ctx context.Context, id int64, dur time.Duration) error {
    return s.users.GrantPro(ctx, id, dur)
}

// DowngradeExpired flips every pro user whose tier_until has passed back
// to free with tier_until = nil. Returns the count of users downgraded.
func (s *Service) DowngradeExpired(ctx context.Context) (int, error) {
    ids, err := s.users.ListExpired(ctx, s.now())

package scheduler

import (
    "context"
    "log/slog"
    "time"

    "github.com/jonboulle/clockwork"
)

// Downgrader is the surface ExpirySweeper needs from users.Service.
type Downgrader interface {
    DowngradeExpired(ctx context.Context) (int, error)
}

// ExpirySweeper runs Downgrader.DowngradeExpired every Interval.
type ExpirySweeper struct {
    down     Downgrader
    interval time.Duration
    clock    clockwork.Clock
    logger   *slog.Logger
}

// NewExpirySweeper returns a sweeper with sensible defaults.
func NewExpirySweeper(d Downgrader, interval time.Duration, clock clockwork.Clock, l *slog.Logger) *ExpirySweeper {
    if interval == 0 { interval = time.Hour }
    if clock == nil { clock = clockwork.NewRealClock() }
    if l == nil { l = slog.Default() }
    return &ExpirySweeper{down: d, interval: interval, clock: clock, logger: l}
}

// Run blocks until ctx is cancelled.
func (s *ExpirySweeper) Run(ctx context.Context) error {
    t := s.clock.NewTicker(s.interval); defer t.Stop()
    for {
        select {
        case <-ctx.Done(): return ctx.Err()
        case <-t.Chan():
            n, err := s.down.DowngradeExpired(ctx)
            if err != nil {
                s.logger.WarnContext(ctx, "expiry sweep", slog.Any("err", err))
                continue
            }
            if n > 0 {
                s.logger.InfoContext(ctx, "expiry sweep", slog.Int("downgraded", n))
            }
        }
    }
}

// (false on UNIQUE(provider, provider_ref) conflict). Idempotent against
// retried webhooks.
func (r *SubscriptionsRepo) InsertIfAbsent(ctx context.Context, in SubscriptionInsert) (int64, bool, error) {

        INSERT INTO subscriptions (user_id, plan, provider, provider_ref, amount, currency, started_at, expires_at, status)

        ON CONFLICT (provider, provider_ref) DO NOTHING

        in.UserID, in.Plan, in.Provider, in.ProviderRef,
        in.Amount, in.Currency, in.StartedAt, in.ExpiresAt, in.Status,

        if gerr != nil { return 0, false, fmt.Errorf("conflict lookup: %w", gerr) }
        return existing.ID, false, nil
    }
    if err != nil { return 0, false, fmt.Errorf("insert subscription: %w", err) }
    return id, true, nil
}

// SubscriptionWriter is the surface the orchestrator needs from
// SubscriptionsRepo.
type SubscriptionWriter interface {
    InsertIfAbsent(ctx context.Context, in postgres.SubscriptionInsert) (int64, bool, error)
}

// UserGranter wraps the users service.
type UserGranter interface {
    GrantPro(ctx context.Context, userDBID int64, dur time.Duration) error
}

// UserResolver translates TG user id → internal DB id (RegisterOnStart
// handles the create-if-missing case).
type UserResolver interface {
    RegisterOnStart(ctx context.Context, tgID int64, username string) (postgres.User, bool, error)
}

// Orchestrator turns an Activation into a persisted subscription + a Pro
// grant. The whole operation is idempotent: a repeated webhook for the same
// provider_ref does not duplicate the grant.
type Orchestrator struct {
    subs    SubscriptionWriter
    granter UserGranter
    users   UserResolver
    logger  *slog.Logger
}

func NewOrchestrator(s SubscriptionWriter, g UserGranter, u UserResolver, l *slog.Logger) *Orchestrator {
    if l == nil { l = slog.Default() }
    return &Orchestrator{subs: s, granter: g, users: u, logger: l}
}

// Activate persists the subscription and grants Pro if and only if the
// subscription row is new. Returns whether the call resulted in a grant
// (false when this provider_ref was already processed).
func (o *Orchestrator) Activate(ctx context.Context, a Activation) (bool, error) {
    u, _, err := o.users.RegisterOnStart(ctx, a.UserTGID, "")
    if err != nil { return false, fmt.Errorf("resolve user: %w", err) }
    now := time.Now()
    _, inserted, err := o.subs.InsertIfAbsent(ctx, postgres.SubscriptionInsert{
        UserID: u.ID, Plan: a.Plan, Provider: a.Provider, ProviderRef: a.ProviderRef,
        Amount: a.Amount, Currency: a.Currency,
        StartedAt: now, ExpiresAt: now.Add(a.Period),
        Status: "active",
    })
    if err != nil { return false, fmt.Errorf("insert subscription: %w", err) }
    if !inserted {
        o.logger.InfoContext(ctx, "duplicate payment ignored",
            slog.String("provider_ref", a.ProviderRef))
        return false, nil
    }
    if err := o.granter.GrantPro(ctx, u.ID, a.Period); err != nil {
        return false, fmt.Errorf("grant pro: %w", err)
    }
    return true, nil
}

    Insert(ctx context.Context, userID int64, name string, emb pgvector.Vector) (int64, error)
    ListByUser(ctx context.Context, userID int64) ([]postgres.UserTopic, error)
    Count(ctx context.Context, userID int64) (int, error)
    Delete(ctx context.Context, userID int64, name string) error

}

// Embedder is the slice of llm.Provider the service uses to embed user
// topic names. ALWAYS called only on AddTopic — never per-digest — so
// INV-5 (Free user latency budget) is preserved.
type Embedder interface {
    Embed(ctx context.Context, req llm.EmbedRequest) (llm.EmbedResponse, error)
}

type Service struct {
    repo  TopicsRW
    emb   Embedder
    model string
}

func NewService(repo TopicsRW, emb Embedder, model string) *Service {
    return &Service{repo: repo, emb: emb, model: model}
}

// AddTopic embeds name and persists it for userID. Caller is responsible
// for verifying userID is Pro.
func (s *Service) AddTopic(ctx context.Context, userID int64, name string) error {
    n, err := s.repo.Count(ctx, userID); if err != nil { return fmt.Errorf("count: %w", err) }
    if n >= MaxTopicsPerUser { return ErrTopicLimitReached }
    resp, err := s.emb.Embed(ctx, llm.EmbedRequest{Model: s.model, Texts: []string{name}})
    if err != nil { return fmt.Errorf("embed: %w", err) }
    if len(resp.Vectors) != 1 { return fmt.Errorf("empty embedding") }
    if _, err := s.repo.Insert(ctx, userID, name, pgvector.NewVector(resp.Vectors[0])); err != nil {
        return fmt.Errorf("persist: %w", err)
    }
    return nil
}

// RemoveTopic deletes the named topic.
func (s *Service) RemoveTopic(ctx context.Context, userID int64, name string) error {
    return s.repo.Delete(ctx, userID, name)
}

// ListTopics returns just the names.
func (s *Service) ListTopics(ctx context.Context, userID int64) ([]string, error) { ... }

// userID's custom topics within the default similarity threshold.
func (s *Service) MatchesAny(ctx context.Context, userID int64, centroid pgvector.Vector) (bool, error) {
    return s.repo.MatchesAny(ctx, userID, centroid, MatchDistance)

type AlertThrottle struct{ rdb *redis.Client }

func NewAlertThrottle(c *Client) *AlertThrottle { return &AlertThrottle{rdb: c.RDB()} }

// Allow attempts to acquire the throttle slot for (userID, topic) with TTL
// throttle. Returns true on acquisition, false if already throttled.
func (t *AlertThrottle) Allow(ctx context.Context, userID int64, topic string, throttle time.Duration) (bool, error) {
    key := fmt.Sprintf("alert_throttle:%d:%s", userID, topic)
    ok, err := t.rdb.SetNX(ctx, key, "1", throttle).Result()
    if err != nil { return false, fmt.Errorf("setnx: %w", err) }
    return ok, nil
}

// breaking news to Pro users whose alert_threshold has been crossed. Free
// users are skipped before any LLM-charging work happens (INV-2).
package alerts

import (
    "context"
    "encoding/json"
    "fmt"
    "log/slog"
    "time"

    "github.com/virsi/mute-bot/internal/storage/postgres"

// UsersForAlert is the slice the worker needs from postgres.UsersRepo +
// SettingsRepo. ListEligible returns Pro users with alerts_enabled = true,
// joined with their alert_threshold and alert_throttle_minutes.
type UsersForAlert interface {
    ListEligible(ctx context.Context) ([]EligibleUser, error)
}

// EligibleUser bundles every field the worker needs to make a decision in
// one query, avoiding N+1 lookups.
type EligibleUser struct {
    UserID         int64
    TGUserID       int64
    AlertThreshold int
    ThrottleMin    int
}

// Throttler decides whether a (user, topic) push is allowed right now.
type Throttler interface {
    Allow(ctx context.Context, userID int64, topic string, throttle time.Duration) (bool, error)
}

// ClusterReader is the slice of ClustersRepo needed to render a cluster
// into a one-line alert.
type ClusterReader interface {
    Get(ctx context.Context, id int64) (postgres.Cluster, error)
    SetScore(ctx context.Context, id int64, score float32) error // unused; kept for symmetry
}

// Sender is the surface used to push the alert.
type Sender interface {
    SendDigest(ctx context.Context, tgUserID int64, text string) error
}

// Event is the cluster.scored payload shape.
type Event struct {
    ClusterID int64    `json:"cluster_id"`
    Score     float32  `json:"score"`
    Topics    []string `json:"topics"`
}

// Worker handles cluster.scored events.
type Worker struct {
    users     UsersForAlert
    clusters  ClusterReader
    throttle  Throttler
    sender    Sender
    logger    *slog.Logger
}

func NewWorker(u UsersForAlert, c ClusterReader, t Throttler, s Sender, l *slog.Logger) *Worker {
    if l == nil { l = slog.Default() }
    return &Worker{users: u, clusters: c, throttle: t, sender: s, logger: l}
}

// Handle is the JetStream callback. It enumerates eligible Pro users, runs
// per-user threshold + throttle, and sends. Free users are never enumerated.
func (w *Worker) Handle(ctx context.Context, payload []byte) error {
    var ev Event
    if err := json.Unmarshal(payload, &ev); err != nil { return fmt.Errorf("unmarshal: %w", err) }
    cluster, err := w.clusters.Get(ctx, ev.ClusterID); if err != nil { return fmt.Errorf("get cluster: %w", err) }

    users, err := w.users.ListEligible(ctx); if err != nil { return fmt.Errorf("list eligible: %w", err) }
    text := fmt.Sprintf("🚨 %s\n\n%s", cluster.Headline, cluster.Summary)
    topicKey := firstNonEmpty(ev.Topics)

    for _, u := range users {
        if int(ev.Score*100) < u.AlertThreshold { continue }
        ok, err := w.throttle.Allow(ctx, u.UserID, topicKey, time.Duration(u.ThrottleMin)*time.Minute)
        if err != nil {
            w.logger.WarnContext(ctx, "throttle", slog.Any("err", err)); continue
        }
        if !ok { continue }
        if err := w.sender.SendDigest(ctx, u.TGUserID, text); err != nil {
            w.logger.WarnContext(ctx, "send alert", slog.Int64("user", u.UserID), slog.Any("err", err))
        }
    }
    return nil
}

func firstNonEmpty(s []string) string { for _, x := range s { if x != "" { return x } }; return "other" }

// ListAlertEligible returns (user_id, tg_user_id, alert_threshold,
// alert_throttle_minutes) for every non-blocked Pro user with alerts_enabled.
type AlertEligible struct {
    UserID, TGUserID int64
    Threshold, ThrottleMin int
}

func (r *UsersRepo) ListAlertEligible(ctx context.Context) ([]AlertEligible, error) {
    const q = `
        SELECT u.id, u.tg_user_id, s.alert_threshold, s.alert_throttle_minutes
        FROM users u
        JOIN user_settings s ON s.user_id = u.id
        WHERE u.blocked = false
          AND u.tier = 'pro'
          AND (u.tier_until IS NULL OR u.tier_until > now())
          AND s.alerts_enabled = true`
    rows, err := r.p.Pool().Query(ctx, q); if err != nil { return nil, fmt.Errorf("list eligible: %w", err) }
    defer rows.Close()
    var out []AlertEligible
    for rows.Next() {
        var e AlertEligible
        if err := rows.Scan(&e.UserID, &e.TGUserID, &e.Threshold, &e.ThrottleMin); err != nil { return nil, err }
        out = append(out, e)
    }
    return out, rows.Err()
}
